\section{Introduction}

\subsection{Purpose}
This Software Requirements Specification (SRS) document describes the complete functional and non-functional requirements for \textbf{Smart Ledger}, a cloud-based, cross-platform financial management application. It serves as the formal agreement between stakeholders (clients, developers, testers, and project managers) and defines what the system must do before any implementation decisions are finalized.

\subsection{Scope}
Smart Ledger is a premium financial management platform targeting small-to-medium enterprises (SMEs) and professional contractors. It replaces fragmented spreadsheets and manual ledger books with a unified, automated system. The platform covers:

\begin{itemize}[leftmargin=*]
\item User authentication and role-based access
\item Client and account receivable management
\item Cashbook and unified ledger (double-entry bookkeeping)
\item Transaction entry, categorization, and batch upload
\item Financial reporting (Balance Sheet, P\&L, Cash Flow)
\item Settings, user management, and system configuration
\end{itemize}

\textbf{Out of Scope (v1.0):} AI-powered forecasting, blockchain integration, payroll module, intercompany settlement automation.

\subsection{Objectives}
\begin{itemize}[leftmargin=*]
\item Reduce manual bookkeeping overhead by 70–80\%
\item Achieve 99.9\% accuracy in transaction recording
\item Provide real-time financial dashboards across Web, Android, iOS, and Desktop
\item Ensure enterprise-grade security (encryption, MFA, RBAC)
\item Enable month-end close in 1–2 days instead of 5–10 days
\end{itemize}

\section{Overall Description}

\subsection{Product Perspective}
Smart Ledger is a standalone cloud-native SaaS application. It does not replace an entire ERP system but integrates with third-party services (banks, payment gateways, CRMs) via REST APIs. All client data is stored in a Supabase-managed PostgreSQL database with Row-Level Security to ensure full multi-tenant data isolation.

\subsubsection{Authentication Module (FR-AUTH)}

\subsubsection{FR-AUTH-01: User Registration}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Full name, email address, password, organization name
\item \textbf{Processing:} Validate email format; enforce password strength (min 8 chars, 1 uppercase, 1 number, 1 special character); create user record in `auth.users` via Supabase Auth; create corresponding profile in `public.profiles`
\item \textbf{Output:} Confirmation email sent; user redirected to onboarding screen
\item \textbf{Conditions:} Email must be unique in the system
\end{itemize}

\subsubsection{FR-AUTH-02: User Login}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Email address, password
\item \textbf{Processing:} Validate credentials via Supabase Auth; generate JWT token; load user role and permissions; initialize session
\item \textbf{Output:} User redirected to role-appropriate dashboard
\item \textbf{Conditions:} If credentials are invalid, display error "Invalid email or password." Lock account after 5 consecutive failures for 15 minutes
\end{itemize}

\subsubsection{FR-AUTH-03: Multi-Factor Authentication (MFA)}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} TOTP code from authenticator app (Google Authenticator / Authy)
\item \textbf{Processing:} Verify TOTP against stored secret; extend session on success
\item \textbf{Output:} Session fully authenticated; access granted
\item \textbf{Conditions:} MFA is optional during beta; admin can enforce MFA org-wide
\end{itemize}

\subsubsection{FR-AUTH-04: Password Reset}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Registered email address
\item \textbf{Processing:} Send password reset link via Supabase Auth email; link expires in 60 minutes
\item \textbf{Output:} User can set a new password; old sessions invalidated
\item \textbf{Conditions:} Link is single-use
\end{itemize}

\subsubsection{FR-AUTH-05: Session Management}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} User activity / inactivity
\item \textbf{Processing:} Auto-logout after 30 minutes of inactivity; refresh JWT token on active use
\item \textbf{Output:} Session terminated; user redirected to login page with notification
\item \textbf{Conditions:} User is warned 2 minutes before auto-logout
\end{itemize}

\subsubsection{FR-AUTH-06: Logout}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} User clicks "Logout"
\item \textbf{Processing:} Invalidate JWT token via Supabase Auth sign-out; clear local session state
\item \textbf{Output:} User redirected to login screen; all cached data cleared
\end{itemize}

\subsubsection{Dashboard Module (FR-DASH)}

\subsubsection{FR-DASH-01: KPI Overview Widgets}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Authenticated user session, selected date range
\item \textbf{Processing:} Query aggregated financial data (net revenue, total expenses, cash on hand, outstanding receivables); compute KPIs in real-time
\item \textbf{Output:} Display widgets: Net Revenue, Total Expenses, Cash Balance, Receivables, Payables
\item \textbf{Conditions:} Data refreshes every 60 seconds or on WebSocket push event
\end{itemize}

\subsubsection{FR-DASH-02: Revenue vs. Expense Trend Chart}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date range filter (default: current month)
\item \textbf{Processing:} Aggregate daily/weekly/monthly transaction data grouped by type (income/expense); format for fl\_chart line/bar chart
\item \textbf{Output:} Interactive line chart showing revenue and expense trends
\item \textbf{Conditions:} Chart must render within 2 seconds; supports zoom/pan
\end{itemize}

\subsubsection{FR-DASH-03: Recent Transactions Feed}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Authenticated session
\item \textbf{Processing:} Fetch last 10 transactions sorted by date descending
\item \textbf{Output:} Scrollable list showing: date, description, amount, category, status
\item \textbf{Conditions:} Clicking a transaction opens the transaction detail screen
\end{itemize}

\subsubsection{FR-DASH-04: Alerts \& Notifications}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} System event triggers (negative cash balance, overdue invoice, budget breach)
\item \textbf{Processing:} Evaluate alert conditions on data update; push notification to user
\item \textbf{Output:} In-app notification banner and notification center entry
\item \textbf{Conditions:} Each alert type can be enabled/disabled per user preference
\end{itemize}

\subsubsection{FR-DASH-05: Dashboard Customization}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} User drag-and-drop or toggle widget visibility
\item \textbf{Processing:} Save widget layout preferences to `public.user\_preferences` table
\item \textbf{Output:} Customized dashboard layout persists across sessions and devices
\item \textbf{Conditions:} Minimum 2 widgets must remain visible at all times
\end{itemize}

\subsubsection{Client \& Account Receivable Management (FR-CL)}

\subsubsection{FR-CL-01: Add New Client}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Client name, email, phone, address, tax ID, payment terms, credit limit, currency
\item \textbf{Processing:} Validate required fields; check for duplicate email/tax ID within organization; insert record into `public.clients`
\item \textbf{Output:} Client profile created; appears in client list
\item \textbf{Conditions:} Client name and email are mandatory
\end{itemize}

\subsubsection{FR-CL-02: Edit Client Profile}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Updated client fields
\item \textbf{Processing:} Validate changes; update `public.clients` record; log change in audit trail
\item \textbf{Output:} Updated client profile saved; success toast shown
\item \textbf{Conditions:} Only Accountant role and above can edit clients
\end{itemize}

\subsubsection{FR-CL-03: View Client Outstanding Balance}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Selected client
\item \textbf{Processing:} Sum all unpaid invoices and credits for client; compute aging schedule (0–30, 31–60, 61–90, 90+ days)
\item \textbf{Output:} Balance summary card + aging schedule table
\item \textbf{Conditions:} Balance updates in real-time when new transactions are posted
\end{itemize}

\subsubsection{FR-CL-04: Generate Invoice}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Client ID, line items (description, quantity, unit price), due date, tax rate
\item \textbf{Processing:} Calculate subtotal, tax, and total; assign invoice number (auto-incremented); insert into `public.invoices` and `public.invoice\_items`
\item \textbf{Output:} Formatted PDF invoice generated; option to email directly to client
\item \textbf{Conditions:} Invoice number format: `INV-YYYY-XXXXXX`
\end{itemize}

\subsubsection{FR-CL-05: Track Invoice Status}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Invoice ID
\item \textbf{Processing:} Check payment records against invoice amount; update status (Draft → Sent → Partial → Paid → Overdue)
\item \textbf{Output:} Invoice status badge updated; overdue invoices trigger alert
\item \textbf{Conditions:} Status "Overdue" triggered automatically when due date passes and balance > 0
\end{itemize}

\subsubsection{FR-CL-06: Delete Client}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Client ID, admin confirmation
\item \textbf{Processing:} Soft-delete client (set `is\_active = false`); retain all historical transactions and invoices for audit
\item \textbf{Output:} Client removed from active list; data preserved
\item \textbf{Conditions:} Only Administrator can delete clients; cannot delete clients with unpaid invoices
\end{itemize}

\subsubsection{Cashbook Module (FR-CB)}

\subsubsection{FR-CB-01: View Cashbook}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date range filter, account filter
\item \textbf{Processing:} Fetch all cash receipts and payments in selected range; calculate running balance after each entry
\item \textbf{Output:} Tabular view with columns: Date, Reference, Description, Debit, Credit, Balance
\item \textbf{Conditions:} Sorted by date ascending; supports pagination (50 rows per page)
\end{itemize}

\subsubsection{FR-CB-02: Add Cash Receipt}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date, reference number, description, amount, category, client/source
\item \textbf{Processing:} Validate amount > 0; insert debit entry into `public.cashbook\_entries`; auto-post corresponding credit to ledger account
\item \textbf{Output:} Entry appears in cashbook; running balance updated
\item \textbf{Conditions:} Cannot post to a closed accounting period
\end{itemize}

\subsubsection{FR-CB-03: Add Cash Payment}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date, reference number, description, amount, category, payee/vendor
\item \textbf{Processing:} Validate amount > 0 and sufficient balance; insert credit entry; auto-post debit to appropriate expense account
\item \textbf{Output:} Entry in cashbook; balance reduced
\item \textbf{Conditions:} Warning shown if payment causes cash balance to go negative
\end{itemize}

\subsubsection{FR-CB-04: Edit Cashbook Entry}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Entry ID, updated fields
\item \textbf{Processing:} Create reversal entry for original; post corrected entry; log both in audit trail
\item \textbf{Output:} Cashbook reflects correction; audit trail shows original and revised entries
\item \textbf{Conditions:} Only Finance Manager and above can edit posted entries
\end{itemize}

\subsubsection{FR-CB-05: Delete/Void Cashbook Entry}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Entry ID, reason for void
\item \textbf{Processing:} Mark entry as `voided`; create reversal posting; log void reason and user
\item \textbf{Output:} Entry marked voided (not deleted); ledger balance corrected
\item \textbf{Conditions:} Voiding requires Finance Manager approval; physical deletion is not permitted
\end{itemize}

\vspace{1em}\hrule\vspace{1em}

\subsubsection{Ledger Module (FR-LG)}

\subsubsection{FR-LG-01: View General Ledger}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Account filter, date range
\item \textbf{Processing:} Fetch all journal entries for selected accounts in range; compute account balance
\item \textbf{Output:} Ledger view: Date, Journal Ref, Description, Debit, Credit, Running Balance per account
\item \textbf{Conditions:} Opening balance shown at top of each account section
\end{itemize}

\subsubsection{FR-LG-02: Create Manual Journal Entry}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date, journal reference, description, debit lines (account, amount), credit lines (account, amount)
\item \textbf{Processing:} Validate total debits == total credits (balanced entry); insert into `public.journal\_entries` and `public.journal\_lines`; update account balances
\item \textbf{Output:} Journal entry posted; ledger accounts updated
\item \textbf{Conditions:} Unbalanced entries are rejected with error: "Debits must equal credits"
\end{itemize}

\subsubsection{FR-LG-03: Chart of Accounts Management}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Account name, account type (Asset/Liability/Equity/Revenue/Expense), account code, parent account
\item \textbf{Processing:} Validate unique account code; insert into `public.accounts`; support hierarchical structure
\item \textbf{Output:} New account appears in chart of accounts; available for journal entries
\item \textbf{Conditions:} Only Administrator can create/delete accounts
\end{itemize}

\subsubsection{FR-LG-04: Account Balance Summary}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} As-of date
\item \textbf{Processing:} Compute closing balance for all accounts as of selected date
\item \textbf{Output:} Trial balance report: Account | Debit Balance | Credit Balance
\item \textbf{Conditions:} Total debits must equal total credits; imbalance flagged as system error
\end{itemize}

\subsubsection{Transactions Module (FR-TX)}

\subsubsection{FR-TX-01: Add Single Transaction}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date, type (Income/Expense/Transfer), amount, category, description, client/vendor, payment method, attachment
\item \textbf{Processing:} Validate all required fields; classify category; insert into `public.transactions`; propagate to cashbook and ledger
\item \textbf{Output:} Transaction recorded; dashboard KPIs updated
\item \textbf{Conditions:} Amount must be > 0; date cannot be in the future unless it is a scheduled transaction
\end{itemize}

\subsubsection{FR-TX-02: Batch Import Transactions}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} CSV/Excel file upload
\item \textbf{Processing:} Parse file; validate each row (required fields, data types, duplicates); preview rows with errors highlighted; on confirm, insert valid rows; log rejected rows
\item \textbf{Output:} Import summary: X records imported, Y records failed; downloadable error report
\item \textbf{Conditions:} Max file size 10 MB; max 5,000 rows per import
\end{itemize}

\subsubsection{FR-TX-03: Edit Transaction}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Transaction ID, updated fields
\item \textbf{Processing:} Check user permission; update record; create audit log entry; propagate changes to cashbook and ledger
\item \textbf{Output:} Transaction updated; audit trail records who changed what and when
\item \textbf{Conditions:} Posted transactions require Finance Manager approval to edit
\end{itemize}

\subsubsection{FR-TX-04: Delete/Void Transaction}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Transaction ID, void reason
\item \textbf{Processing:} Soft-delete; create reversal posting; notify relevant users
\item \textbf{Output:} Transaction voided; balances corrected
\item \textbf{Conditions:} Administrator only; irreversible physical deletion not permitted
\end{itemize}

\subsubsection{FR-TX-05: Filter \& Search Transactions}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Search term, filters (date range, category, amount range, status, client/vendor)
\item \textbf{Processing:} Query `public.transactions` with applied filters; rank by relevance
\item \textbf{Output:} Filtered transaction list; result count displayed
\item \textbf{Conditions:} Response time < 1 second for up to 100,000 records with proper indexing
\end{itemize}

\subsubsection{FR-TX-06: Transaction Categorization}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Transaction description and amount
\item \textbf{Processing:} Match against category rules; suggest most likely category based on history patterns
\item \textbf{Output:} Pre-filled category field; user can override
\item \textbf{Conditions:} Suggestion accuracy target ≥ 80\% after 50+ historical transactions
\end{itemize}

\subsection{Reports Module (FR-RP)}

\subsubsection{FR-RP-01: Profit \& Loss Statement}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date range, comparison period (optional)
\item \textbf{Processing:} Aggregate revenue and expense accounts; compute gross profit, operating profit, net profit; optionally compare with prior period
\item \textbf{Output:} Formatted P\&L report with section totals and variance percentages
\item \textbf{Conditions:} Generated within 5 seconds for up to 1M transactions
\end{itemize}

\subsubsection{FR-RP-02: Balance Sheet}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} As-of date
\item \textbf{Processing:} Compute total assets, liabilities, and equity as of selected date; verify Assets = Liabilities + Equity
\item \textbf{Output:} Standard balance sheet format compliant with GAAP/IFRS
\item \textbf{Conditions:} Any imbalance flagged as a critical error requiring admin review
\end{itemize}

\subsubsection{FR-RP-03: Cash Flow Statement}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Date range
\item \textbf{Processing:} Categorize cash movements into Operating, Investing, Financing activities; compute net cash change
\item \textbf{Output:} Cash flow statement with opening and closing cash balances
\item \textbf{Conditions:} Must reconcile with cashbook closing balance
\end{itemize}

\subsubsection{FR-RP-04: Accounts Receivable Aging Report}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} As-of date
\item \textbf{Processing:} Group outstanding invoices by age bucket: Current, 1–30, 31–60, 61–90, 90+ days
\item \textbf{Output:} Aging table per client with total outstanding
\item \textbf{Conditions:} Highlights clients with 90+ day overdue amounts in red
\end{itemize}

\subsubsection{FR-RP-05: Export Reports}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Report type, format (PDF/Excel/CSV)
\item \textbf{Processing:} Render report data; format per selected type; generate download link
\item \textbf{Output:} Downloadable file delivered to user
\item \textbf{Conditions:} PDF exports include company logo and formatted headers; Excel exports include raw data and formulas
\end{itemize}

\subsubsection{FR-RP-06: Scheduled Reports}
\begin{itemize}[leftmargin=*]
\item \textbf{Input:} Report type, schedule (daily/weekly/monthly), recipient email list
\item \textbf{Processing:} Generate report on schedule; email to recipients; log delivery
\item \textbf{Output:} Report delivered to configured recipients on time
\item \textbf{Conditions:} Failed deliveries retry 3 times; admin notified on persistent failure
\end{itemize}

\subsubsection{FR-ST-04: Audit Log View}
\begin{itemize}[leftmargin=*]

\end{itemize}

\textbf{Role Enforcement} & DELETE operations on financial tables restricted to users with `role = 'administrator'` \\ \hline
\textbf{Audit Log Protection} & `public.audit\_logs` is INSERT-only; no UPDATE or DELETE permitted by any role \\ \hline
\textbf{Client Portal} & External clients can only SELECT their own invoice records \\ \hline
\end{longtable}

\subsection{Glossary}

\begin{longtable}{|p{0.47\textwidth}|p{0.47\textwidth}|}
\hline
Term & Definition \\ \hline
\hline
\textbf{Double-Entry Bookkeeping} & An accounting system where every transaction has equal debit and credit entries \\ \hline
\textbf{Trial Balance} & A summary listing all accounts and their balances; total debits must equal total credits \\ \hline
\textbf{Aging Schedule} & A report grouping outstanding receivables by how overdue they are \\ \hline
\textbf{Chart of Accounts} & A structured list of all financial accounts used by an organization \\ \hline
\textbf{Soft Delete} & Marking a record as inactive/voided without removing it from the database \\ \hline
\textbf{Multi-Tenancy} & A single application instance serving multiple organizations with full data isolation \\ \hline
\textbf{RLS} & Row-Level Security — a PostgreSQL feature restricting data access at the row level based on user context \\ \hline
\textbf{JWT} & JSON Web Token — a signed token used to authenticate API requests \\ \hline
\textbf{TOTP} & Time-based One-Time Password — used for MFA \\ \hline
\textbf{Fiscal Year} & A 12-month accounting period used for financial reporting \\ \hline
\textbf{Reconciliation} & The process of ensuring two sets of records (e.g., cashbook and bank statement) agree \\ \hline
\end{longtable}
